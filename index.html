Browse 254 flags and discover facts about countries and territories around the globe

countryData = FileAttachment("countries.json").json()

// Combine every flag file in w2560/ with its matching country details
flagOptions = flag_files
  .map(file => {
    const code = file.replace(/\.png$/i, "");
    const info = countryData[code] || {};
    return {
      file,
      code,
      name: info.name || code.toUpperCase(),
      capital: info.capital || "N/A",
      region: info.region || "N/A",
      population: info.population || "N/A",
      languages: info.languages || "N/A",
      currency: info.currency || "N/A",
      funFact: info.funFact || ""
    };
  })
  .sort((a, b) => a.name.localeCompare(b.name))

// Dropdown menu sorted alphabetically by full country name
viewof selected = Inputs.select(flagOptions, {
  label: "Choose a country or territory:",
  format: d => d.name,
  value: flagOptions.find(d => d.file === "us.png") || flagOptions[0]
})
