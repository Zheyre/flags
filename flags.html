Browse 254 flags from countries and territories around the globe

flagOptions = {
  const regionNames = new Intl.DisplayNames(["en"], { type: "region" });
  const customNames = {
    "gb-eng": "England",
    "gb-nir": "Northern Ireland",
    "gb-sct": "Scotland",
    "gb-wls": "Wales"
  };

  return flag_files
    .map(file => {
      const code = file.replace(/\.png$/i, "");
      const name = customNames[code] || regionNames.of(code.toUpperCase()) || code.toUpperCase();
      return { file, code, name };
    })
    .sort((a, b) => a.name.localeCompare(b.name));
}

// Dropdown menu sorted by full country name
viewof selected = Inputs.select(flagOptions, {
  label: "Choose a country or territory:",
  format: d => d.name,
  value: flagOptions.find(d => d.file === "us.png") || flagOptions[0]
})

// Rounded card showing the selected flag and full country name underneath
html`<div class="flag-card" key="${selected.file}">
  <div class="flag-image-wrapper">
    <img
      class="flag-image"
      src="w2560/${selected.file}"
      alt="Flag of ${selected.name}"
    />
  </div>
  <h2 class="country-name">${selected.name}</h2>
</div>`
